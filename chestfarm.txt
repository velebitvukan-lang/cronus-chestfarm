local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer

local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

local statusMsg = Instance.new("Message")
statusMsg.Text = "CRONUS v2 loading..."
statusMsg.Parent = Workspace

local function notify(t)
    pcall(function() statusMsg.Text = "[Cronus] " .. tostring(t) end)
    print("[Cronus]", tostring(t))
end

local Settings = {
    FarmEnabled = true,
    Noclip = true,
    Teleport = true,
    WalkSpeed = 70,
    ArriveDistance = 6,
    ScanInterval = 1.0,
    InteractCooldown = 0.3,
    AvoidVoidY = -50,
    ChestNames = { "chest", "crate", "loot", "reward", "supply", "bounty", "bandit", "combat", "collect", "gift", "present" },
    FilterItems = { "no3 sawed off", "clue scroll" },
}

local State = {
    Chests = {},
    LastScan = 0,
    LastInteract = 0,
    AllRemotes = {},
    OpenRemotes = {},
    ClaimRemotes = {},
    SampleChestNames = {},
}

local function isInteractive(obj)
    if obj:FindFirstChildWhichIsA("ClickDetector") then return true end
    if obj:FindFirstChildWhichIsA("ProximityPrompt") then return true end
    return false
end

local function chestNameHit(name)
    local n = string.lower(name)
    for _, frag in ipairs(Settings.ChestNames) do
        if string.find(n, frag) then return true end
    end
    return false
end

local function isChest(obj)
    if chestNameHit(obj.Name) then
        local pos = nil
        if obj:IsA("BasePart") then pos = obj.Position
        elseif obj:IsA("Model") then
            if obj.PrimaryPart then pos = obj.PrimaryPart.Position
            else
                local p = obj:FindFirstChildWhichIsA("BasePart")
                if p then pos = p.Position end
            end
        else
            local p = obj:FindFirstChildWhichIsA("BasePart")
            if p then pos = p.Position end
        end
        if not pos then return false end
        if pos.Y < Settings.AvoidVoidY then return false end
        return true, pos
    end
    if isInteractive(obj) then
        local pos = nil
        if obj:IsA("BasePart") then pos = obj.Position
        elseif obj:IsA("Model") then
            if obj.PrimaryPart then pos = obj.PrimaryPart.Position
            else
                local p = obj:FindFirstChildWhichIsA("BasePart")
                if p then pos = p.Position end
            end
        else
            local p = obj:FindFirstChildWhichIsA("BasePart")
            if p then pos = p.Position end
        end
        if not pos then return false end
        if pos.Y < Settings.AvoidVoidY then return false end
        return true, pos
    end
    return false
end

local function scanWorld()
    local results = {}
    local samples = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        local ok, pos = isChest(obj)
        if ok then
            table.insert(results, { Object = obj, Position = pos })
            if #samples < 12 then
                table.insert(samples, obj.Name)
            end
        end
    end
    table.insert(State.SampleChestNames, "lastscan:" .. #results)
    for _, s in ipairs(samples) do
        table.insert(State.SampleChestNames, s)
    end
    if #State.SampleChestNames > 30 then
        for i = 1, (#State.SampleChestNames - 30) do
            table.remove(State.SampleChestNames, 1)
        end
    end
    return results
end

local function discoverRemotes()
    local all = {}
    local opens = {}
    local claims = {}
    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") then
            table.insert(all, obj.Name)
            local n = string.lower(obj.Name)
            if string.find(n, "claim") or string.find(n, "take") or string.find(n, "give") or string.find(n, "collect") or string.find(n, "pickup") or string.find(n, "reward") then
                table.insert(claims, obj)
            elseif string.find(n, "chest") or string.find(n, "loot") or string.find(n, "crate") or string.find(n, "open") or string.find(n, "item") or string.find(n, "inv") or string.find(n, "use") or string.find(n, "interact") or string.find(n, "reward") then
                table.insert(opens, obj)
            end
        end
    end
    State.AllRemotes = all
    State.OpenRemotes = opens
    State.ClaimRemotes = claims
end

local function itemAllowed(name)
    if not name then return false end
    local n = string.lower(name)
    for _, frag in ipairs(Settings.FilterItems) do
        if string.find(n, string.lower(frag)) then return true end
    end
    return false
end

local function travelTo(pos)
    Character = LocalPlayer.Character
    if not Character then return false end
    HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
    if not HumanoidRootPart then return false end
    if Settings.Teleport then
        HumanoidRootPart.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
        return true
    else
        local hum = Character:FindFirstChildOfClass("Humanoid")
        if hum then hum.WalkSpeed = Settings.WalkSpeed end
        HumanoidRootPart.CFrame = HumanoidRootPart.CFrame:Lerp(CFrame.new(pos + Vector3.new(0, 3, 0)), 0.3)
        return true
    end
end

local function tryOpenViaRemotes(obj)
    local fired = false
    for _, r in ipairs(State.OpenRemotes) do
        pcall(function() r:FireServer(obj) end)
        pcall(function() r:FireServer(obj.Position) end)
        pcall(function() r:FireServer(obj.Name) end)
        pcall(function() r:FireServer() end)
        fired = true
    end
    local localRemote = obj:FindFirstChildWhichIsA("RemoteEvent")
    if localRemote then
        pcall(function() localRemote:FireServer() end)
        pcall(function() localRemote:FireServer(obj) end)
        fired = true
    end
    return fired
end

local function openChest(obj)
    local fired = false
    local cd = obj:FindFirstChildWhichIsA("ClickDetector")
    if cd then
        pcall(function() cd:FireServer() end)
        pcall(function() cd.MouseClick:Fire(LocalPlayer) end)
        fired = true
    end
    for _, d in ipairs(obj:GetDescendants()) do
        if d:IsA("ClickDetector") then
            pcall(function() d:FireServer() end)
            pcall(function() d.MouseClick:Fire(LocalPlayer) end)
            fired = true
        end
    end
    local function trig(p)
        pcall(function() p:InputHoldBegin() wait(0.08) p:InputHoldEnd() end)
        pcall(function() p:Trigger() end)
    end
    local pp = obj:FindFirstChildWhichIsA("ProximityPrompt")
    if pp then trig(pp) fired = true end
    for _, d in ipairs(obj:GetDescendants()) do
        if d:IsA("ProximityPrompt") then trig(d) fired = true end
    end
    if tryOpenViaRemotes(obj) then fired = true end
    pcall(function() if obj.Open then obj:Open() end end)
    pcall(function() if obj.Activate then obj:Activate() end end)
    return fired
end

local function claimLoot(chestObj, lootNames)
    if lootNames and #lootNames > 0 then
        for _, item in ipairs(lootNames) do
            if itemAllowed(item) then
                for _, r in ipairs(State.ClaimRemotes) do
                    pcall(function() r:FireServer(item) end)
                    pcall(function() r:FireServer(chestObj, item) end)
                end
            end
        end
    else
        for _, r in ipairs(State.ClaimRemotes) do
            pcall(function() r:FireServer(chestObj) end)
            pcall(function() r:FireServer() end)
        end
    end
end

local function readChestLoot(obj)
    local names = {}
    for _, d in ipairs(obj:GetDescendants()) do
        if d:IsA("StringValue") or d:IsA("TextLabel") or d:IsA("StringConstrainedValue") then
            if d.Value and string.len(tostring(d.Value)) > 0 then
                table.insert(names, tostring(d.Value))
            end
        end
    end
    return names
end

local function farmTick()
    local now = tick()
    if now - State.LastScan >= Settings.ScanInterval then
        State.Chests = scanWorld()
        State.LastScan = now
        notify("FARM:" .. (Settings.FarmEnabled and "ON" or "OFF") .. " Chests:" .. #State.Chests .. " Remotes:open=" .. #State.OpenRemotes .. " claim=" .. #State.ClaimRemotes .. " | [P] diag [F]arm")
    end
    if #State.Chests > 0 and (now - State.LastInteract) >= Settings.InteractCooldown then
        local target = State.Chests[1]
        if target and target.Object and target.Object.Parent then
            local ok, pos = isChest(target.Object)
            if ok then
                travelTo(pos)
                openChest(target.Object)
                local loot = readChestLoot(target.Object)
                claimLoot(target.Object, loot)
                notify("opened: " .. target.Object.Name .. " | loot:" .. #loot)
                State.LastInteract = now
            end
        end
        table.remove(State.Chests, 1)
    end
end

local NoclipConn = nil
local function setNoclip(on)
    if NoclipConn then NoclipConn:Disconnect() NoclipConn = nil end
    if on then
        NoclipConn = RunService.Stepped:Connect(function()
            local c = LocalPlayer.Character
            if not c then return end
            for _, part in ipairs(c:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
            local hum = c:FindFirstChildOfClass("Humanoid")
            if hum then hum:ChangeState(11) end
        end)
    else
        local c = LocalPlayer.Character
        if c then
            for _, part in ipairs(c:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

local FarmConn = nil
local function setFarm(on)
    if FarmConn then FarmConn:Disconnect() FarmConn = nil end
    if on then
        FarmConn = RunService.Heartbeat:Connect(function()
            local ok, err = pcall(farmTick)
            if not ok then notify("ERR: " .. tostring(err)) end
        end)
    end
end

LocalPlayer.CharacterAdded:Connect(function(char)
    Character = char
    HumanoidRootPart = char:WaitForChild("HumanoidRootPart")
    if Settings.Noclip then setNoclip(true) end
end)

UserInputService.InputBegan:Connect(function(inp, g)
    if g then return end
    local k = inp.KeyCode
    if k == Enum.KeyCode.F then
        Settings.FarmEnabled = not Settings.FarmEnabled
        setFarm(Settings.FarmEnabled)
        notify("FARM:" .. (Settings.FarmEnabled and "ON" or "OFF") .. " Chests:" .. #State.Chests)
    elseif k == Enum.KeyCode.N then
        Settings.Noclip = not Settings.Noclip
        setNoclip(Settings.Noclip)
        notify("NOCLIP:" .. (Settings.Noclip and "ON" or "OFF"))
    elseif k == Enum.KeyCode.T then
        Settings.Teleport = not Settings.Teleport
        notify("TELEPORT:" .. (Settings.Teleport and "ON" or "OFF"))
    elseif k == Enum.KeyCode.P then
        State.Chests = scanWorld()
        discoverRemotes()
        local diag = "DIAG chests=" .. #State.Chests .. " | samples: "
        for i = 1, math.min(8, #State.SampleChestNames) do
            diag = diag .. State.SampleChestNames[i] .. ", "
        end
        diag = diag .. " | openR: "
        for i = 1, math.min(10, #State.OpenRemotes) do
            diag = diag .. State.OpenRemotes[i].Name .. ","
        end
        notify(diag)
        print("[Cronus DIAG] chests=" .. #State.Chests)
        for _, s in ipairs(State.SampleChestNames) do print("  chestsample:", s) end
        print("[Cronus DIAG] openRemotes=" .. #State.OpenRemotes)
        for _, r in ipairs(State.OpenRemotes) do print("  open:", r.Name) end
        print("[Cronus DIAG] claimRemotes=" .. #State.ClaimRemotes)
        for _, r in ipairs(State.ClaimRemotes) do print("  claim:", r.Name) end
    end
end)

discoverRemotes()
setNoclip(Settings.Noclip)
setFarm(Settings.FarmEnabled)
notify("CRONUS v2 READY. FARM auto-on. [P]=diag [F]arm [N]oclip [T]p")
spawn(function()
    while true do
        discoverRemotes()
        wait(8)
    end
end)
