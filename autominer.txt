local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer

local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

local statusMsg = Instance.new("Message")
statusMsg.Text = "AUTOMINER LOADED v3"
statusMsg.Parent = Workspace

local function notify(t)
    pcall(function() statusMsg.Text = "[MINER] " .. tostring(t) end)
    print("[MINER]", tostring(t))
end

local Settings = {
    FarmEnabled = false,
    Noclip = false,
    Teleport = true,
    WalkSpeed = 70,
    ArriveDistance = 5,
    ScanInterval = 0.6,
    InteractCooldown = 0.25,
    AvoidVoidY = -50,
    OreNames = { "coal", "copper" },
    GemNames = { "gem", "diamond", "ruby", "emerald", "sapphire", "amber", "jade" },
    SellTargetNames = { "sell", "vendor", "shop", "store", "sales", "miner", "orebuyer", "buyer", "deposit", "trader" },
    ToolNames = { "pickaxe", "pick", "stone pickaxe", "iron pickaxe", "gold pickaxe", "copper pickaxe", "coal pickaxe", "drill", "mining tool" },
    OreRemoteGuesses = { "MineRock", "MineOre", "HitRock", "BreakRock", "OreMined", "Mine", "Rock", "Pickaxe", "HitOre", "CollectOre", "DamageRock", "MineNode", "MineOre", "Ore", "MineCoal", "MineCopper" },
    GemRemoteGuesses = { "CollectGem", "PickupGem", "GemPickup", "CollectItem", "Pickup", "GrabGem", "Collect", "PickUp" },
    SellRemoteGuesses = { "SellOre", "SellAll", "SellInventory", "Sell", "StoreOre", "SellItems", "DepositOre", "SellOres", "BankOre" },
    SellWhenFull = true,
    SellThreshold = 25,
}

local State = {
    Ores = {},
    Gems = {},
    LastScan = 0,
    LastInteract = 0,
    OreRemotes = {},
    GemRemotes = {},
    SellRemotes = {},
    SampleOreNames = {},
    MinedCount = 0,
}

local function objPos(obj)
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") then
        if obj.PrimaryPart then return obj.PrimaryPart.Position end
        local p = obj:FindFirstChildWhichIsA("BasePart")
        if p then return p.Position end
    end
    local p = obj:FindFirstChildWhichIsA("BasePart")
    if p then return p.Position end
    return nil
end

local function nameHit(name, list)
    local n = string.lower(name)
    for _, frag in ipairs(list) do
        if string.find(n, frag) then return true end
    end
    return false
end

local function isOre(obj)
    if not nameHit(obj.Name, Settings.OreNames) then return false end
    local pos = objPos(obj)
    if not pos then return false end
    if pos.Y < Settings.AvoidVoidY then return false end
    return true, pos
end

local function isGem(obj)
    if not nameHit(obj.Name, Settings.GemNames) then return false end
    local pos = objPos(obj)
    if not pos then return false end
    if pos.Y < Settings.AvoidVoidY then return false end
    return true, pos
end

local function scanWorld()
    local ores = {}
    local gems = {}
    local samples = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        local oOk, oPos = isOre(obj)
        if oOk then
            table.insert(ores, { Object = obj, Position = oPos })
            if #samples < 14 then table.insert(samples, obj.Name) end
        end
        local gOk, gPos = isGem(obj)
        if gOk then
            table.insert(gems, { Object = obj, Position = gPos })
        end
    end
    State.Ores = ores
    State.Gems = gems
    State.SampleOreNames = samples
end

local function discoverRemotes()
    local ores = {}
    local gems = {}
    local sells = {}
    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") then
            local n = string.lower(obj.Name)
            for _, g in ipairs(Settings.OreRemoteGuesses) do
                if string.find(n, string.lower(g)) then table.insert(ores, obj) break end
            end
            for _, g in ipairs(Settings.GemRemoteGuesses) do
                if string.find(n, string.lower(g)) then table.insert(gems, obj) break end
            end
            for _, g in ipairs(Settings.SellRemoteGuesses) do
                if string.find(n, string.lower(g)) then table.insert(sells, obj) break end
            end
        end
    end
    State.OreRemotes = ores
    State.GemRemotes = gems
    State.SellRemotes = sells
end

local function equipTool()
    local hum = Character and Character:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = Character:FindFirstChildWhichIsA("Tool")
    if not tool and LocalPlayer.Backpack then
        for _, tn in ipairs(Settings.ToolNames) do
            local t = LocalPlayer.Backpack:FindFirstChild(tn, true)
            if t and t:IsA("Tool") then tool = t break end
        end
        if not tool then tool = LocalPlayer.Backpack:FindFirstChildWhichIsA("Tool") end
    end
    if tool then pcall(function() hum:EquipTool(tool) end) end
    return tool
end

local function goTo(pos)
    Character = LocalPlayer.Character
    HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
    if not HumanoidRootPart then return end
    if Settings.Teleport then
        HumanoidRootPart.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
    else
        local hum = Character:FindFirstChildOfClass("Humanoid")
        if hum then hum.WalkSpeed = Settings.WalkSpeed end
        HumanoidRootPart.CFrame = HumanoidRootPart.CFrame:Lerp(CFrame.new(pos + Vector3.new(0, 3, 0)), 0.4)
    end
end

local function mineOre(ore)
    local fired = false
    local cd = ore:FindFirstChildWhichIsA("ClickDetector")
    if cd then
        pcall(function() cd:FireServer() end)
        pcall(function() cd.MouseClick:Fire(LocalPlayer) end)
        fired = true
    end
    for _, d in ipairs(ore:GetDescendants()) do
        if d:IsA("ClickDetector") then
            pcall(function() d:FireServer() end)
            pcall(function() d.MouseClick:Fire(LocalPlayer) end)
            fired = true
        end
    end
    local function trig(p)
        pcall(function() p:InputHoldBegin() wait(0.08) p:InputHoldEnd() end)
        pcall(function() p:Trigger() end)
    end
    local pp = ore:FindFirstChildWhichIsA("ProximityPrompt")
    if pp then trig(pp) fired = true end
    for _, d in ipairs(ore:GetDescendants()) do
        if d:IsA("ProximityPrompt") then trig(d) fired = true end
    end
    local tool = equipTool()
    if tool then
        pcall(function() tool:Activate() end)
        local tr = tool:FindFirstChildWhichIsA("RemoteEvent")
        if tr then
            pcall(function() tr:FireServer(ore) end)
            pcall(function() tr:FireServer() end)
        end
        fired = true
    end
    for _, r in ipairs(State.OreRemotes) do
        pcall(function() r:FireServer(ore) end)
        pcall(function() r:FireServer(ore.Position) end)
        pcall(function() r:FireServer() end)
        fired = true
    end
    return fired
end

local function collectGem(gem)
    goTo(gem.Position)
    local fired = false
    local cd = gem:FindFirstChildWhichIsA("ClickDetector")
    if cd then
        pcall(function() cd:FireServer() end)
        pcall(function() cd.MouseClick:Fire(LocalPlayer) end)
        fired = true
    end
    for _, d in ipairs(gem:GetDescendants()) do
        if d:IsA("ClickDetector") then
            pcall(function() d:FireServer() end)
            pcall(function() d.MouseClick:Fire(LocalPlayer) end)
            fired = true
        end
    end
    local function trig(p)
        pcall(function() p:InputHoldBegin() wait(0.08) p:InputHoldEnd() end)
        pcall(function() p:Trigger() end)
    end
    local pp = gem:FindFirstChildWhichIsA("ProximityPrompt")
    if pp then trig(pp) fired = true end
    for _, d in ipairs(gem:GetDescendants()) do
        if d:IsA("ProximityPrompt") then trig(d) fired = true end
    end
    for _, r in ipairs(State.GemRemotes) do
        pcall(function() r:FireServer(gem) end)
        pcall(function() r:FireServer(gem.Position) end)
        pcall(function() r:FireServer() end)
        fired = true
    end
    return fired
end

local function findSellTarget()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if nameHit(obj.Name, Settings.SellTargetNames) then
            local p = objPos(obj)
            if p then return obj, p end
        end
    end
    return nil, nil
end

local function doSell()
    local tgt, pos = findSellTarget()
    if tgt and pos then
        goTo(pos)
        local cd = tgt:FindFirstChildWhichIsA("ClickDetector")
        if cd then
            pcall(function() cd:FireServer() end)
            pcall(function() cd.MouseClick:Fire(LocalPlayer) end)
        end
        local pp = tgt:FindFirstChildWhichIsA("ProximityPrompt")
        if pp then
            pcall(function() pp:InputHoldBegin() wait(0.1) pp:InputHoldEnd() end)
            pcall(function() pp:Trigger() end)
        end
    end
    for _, r in ipairs(State.SellRemotes) do
        pcall(function() r:FireServer() end)
        pcall(function() r:FireServer("Ore") end)
        pcall(function() r:FireServer("all") end)
        pcall(function() r:FireServer(true) end)
    end
    notify("SOLD inventory (mined " .. State.MinedCount .. ")")
    State.MinedCount = 0
end

local function farmTick()
    local now = tick()
    if now - State.LastScan >= Settings.ScanInterval then
        scanWorld()
        discoverRemotes()
        State.LastScan = now
        notify("FARM:" .. (Settings.FarmEnabled and "ON" or "OFF") .. " Ores:" .. #State.Ores .. " Gems:" .. #State.Gems .. " Mined:" .. State.MinedCount .. "/" .. Settings.SellThreshold .. " | [J] start [N] sell [P]diag")
    end
    if not Settings.FarmEnabled then return end
    if Settings.SellWhenFull and State.MinedCount >= Settings.SellThreshold then
        doSell()
        return
    end
    if now - State.LastInteract >= Settings.InteractCooldown then
        if #State.Gems > 0 then
            local g = State.Gems[1]
            if g and g.Object and g.Object.Parent then
                goTo(g.Position)
                collectGem(g.Object)
                notify("gem picked: " .. g.Object.Name)
                State.LastInteract = now
                table.remove(State.Gems, 1)
                return
            end
        end
        if #State.Ores > 0 then
            local o = State.Ores[1]
            if o and o.Object and o.Object.Parent then
                goTo(o.Position)
                mineOre(o.Object)
                State.MinedCount = State.MinedCount + 1
                notify("mined: " .. o.Object.Name .. " (" .. State.MinedCount .. ")")
                State.LastInteract = now
                table.remove(State.Ores, 1)
            end
        end
    end
end

local NoclipConn = nil
local function setNoclip(on)
    if NoclipConn then NoclipConn:Disconnect() NoclipConn = nil end
    if on then
        NoclipConn = RunService.Stepped:Connect(function()
            local c = LocalPlayer.Character
            if not c then return end
            for _, part in ipairs(c:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
            local hum = c:FindFirstChildOfClass("Humanoid")
            if hum then hum:ChangeState(11) end
        end)
    else
        local c = LocalPlayer.Character
        if c then
            for _, part in ipairs(c:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

local FarmConn = nil
local function setFarm(on)
    if FarmConn then FarmConn:Disconnect() FarmConn = nil end
    if on then
        FarmConn = RunService.Heartbeat:Connect(function()
            local ok, err = pcall(farmTick)
            if not ok then notify("ERR: " .. tostring(err)) end
        end)
    end
end

LocalPlayer.CharacterAdded:Connect(function(char)
    Character = char
    HumanoidRootPart = char:WaitForChild("HumanoidRootPart")
    if Settings.Noclip then setNoclip(true) end
end)

UserInputService.InputBegan:Connect(function(inp, g)
    if g then return end
    local k = inp.KeyCode
    if k == Enum.KeyCode.J then
        Settings.FarmEnabled = not Settings.FarmEnabled
        if Settings.FarmEnabled then
            Settings.Noclip = true
            Settings.Teleport = true
            setNoclip(true)
        else
            setNoclip(false)
        end
        setFarm(Settings.FarmEnabled)
        notify("FARM:" .. (Settings.FarmEnabled and "ON" or "OFF") .. " NOCLIP:" .. (Settings.FarmEnabled and "ON" or "OFF") .. " TP:ON Ores:" .. #State.Ores)
    elseif k == Enum.KeyCode.T then
        Settings.Teleport = not Settings.Teleport
        notify("TELEPORT:" .. (Settings.Teleport and "ON" or "OFF"))
    elseif k == Enum.KeyCode.N then
        doSell()
    elseif k == Enum.KeyCode.P then
        scanWorld()
        discoverRemotes()
        local smp = ""
        for i = 1, math.min(10, #State.SampleOreNames) do
            smp = smp .. " " .. State.SampleOreNames[i] .. ","
        end
        notify("DIAG ores=" .. #State.Ores .. " gems=" .. #State.Gems .. " sellR=" .. #State.SellRemotes .. " samples:" .. smp)
        print("[MINER DIAG] ores=" .. #State.Ores)
        for _, s in ipairs(State.SampleOreNames) do print("  oresample:", s) end
    end
end)

discoverRemotes()
setNoclip(Settings.Noclip)
notify("AUTOMINER v3 READY. [J]=START/STOP [N]=sell [T]p [P]diag")
spawn(function()
    while true do
        discoverRemotes()
        wait(8)
    end
end)
