local function getHRP()
    local c = LP.Character
    if c then return c:FindFirstChild("HumanoidRootPart") end
    return nil
end

local function tp(hrp, pos)
    pcall(function() hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0)) end)
end

local function hoverAbove(hrp, pos)
    pcall(function()
        hrp.CFrame = CFrame.new(pos + Vector3.new(0, hoverY, 0))
        hrp.Velocity = Vector3.new(0, 0, 0)
    end)
end

local function extendWeapon()
    pcall(function()
        local c = LP.Character
        if not c then return end
        local hrp = c:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local t = c:FindFirstChildOfClass("Tool") or (LP.Backpack and LP.Backpack:FindFirstChildOfClass("Tool"))
        if not t then return end
        local part = t:FindFirstChild("Handle") or t:FindFirstChildOfClass("BasePart")
        if not part then return end
        part.Massless = true
        part.CanCollide = false
        part.Size = Vector3.new(0.6, 0.6, 60)
        part.CFrame = hrp.CFrame * CFrame.new(0, 0, -30)
    end)
end

local function attack(enemy)
    pcall(function()
        local c = LP.Character
        if not c then return end
        local hrp = c:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local human = c:FindFirstChildOfClass("Humanoid")
        local t = c:FindFirstChildOfClass("Tool")
        if not t and LP.Backpack then t = LP.Backpack:FindFirstChildOfClass("Tool") end
        if t then human:EquipTool(t) end
        local eh = enemy and enemy:FindFirstChild("HumanoidRootPart")
        if eh then
            hrp.CFrame = CFrame.new(eh.Position + Vector3.new(0, strikeGap, 0))
            hrp.CFrame = CFrame.lookAt(hrp.Position, eh.Position)
            hrp.Velocity = Vector3.new(0, 0, 0)
        end
        extendWeapon()
        VirtualUser:CaptureController()
        VirtualUser:ClickButton1(Vector2.new(0, 0))
    end)
end

local function fruitMatch(obj)
    if not obj:IsA("Tool") then return false end
    local nm = string.lower(obj.Name)
    for f, _ in pairs(Selected) do
        if Selected[f] and (nm == string.lower(f.."-"..f) or string.find(nm, string.lower(f))) then
            return true
        end
    end
    return false
end

local function findSelectedFruit()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Tool") and not obj:IsDescendantOf(LP.Character) and not obj:IsDescendantOf(LP.Backpack) then
            if fruitMatch(obj) then return obj end
        end
    end
    return nil
end

local function bringFruits(hrp)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Tool") and not obj:IsDescendantOf(LP.Character) and not obj:IsDescendantOf(LP.Backpack) then
            if fruitMatch(obj) then
                pcall(function()
                    if obj:FindFirstChild("Handle") then
                        obj.Handle.CFrame = hrp.CFrame
                    else
                        obj:SetPrimaryPartCFrame(hrp.CFrame)
                    end
                end)
            end
        end
    end
end

local function hop()
    local ok, err = pcall(function()
        local data = HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?sortOrder=Asc&limit=100"))
        if data and data.data and #data.data > 0 then
            local s = data.data[math.random(1, #data.data)]
            TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LP)
        else
            TeleportService:Teleport(game.PlaceId, LP)
        end
    end)
    if not ok then
        pcall(function()
            local ok2 = pcall(function() TeleportService:Teleport(game.PlaceId, LP) end)
            if not ok2 then setStatus("hop fail: " .. tostring(err):sub(1, 40)) end
        end)
    end
end

local function snipe(hrp)
    local fruit = findSelectedFruit()
    if fruit then
        local pos = fruit:FindFirstChild("Handle") and fruit.Handle.Position or fruit:GetPivot().Position
        tp(hrp, pos)
        pcall(function()
            if fruit:FindFirstChild("Handle") then fruit.Handle.CFrame = hrp.CFrame end
        end)
        pcall(function() firetouchinterest(fruit, hrp, 0) firetouchinterest(fruit, hrp, 1) end)
        setStatus("sniped " .. fruit.Name)
    else
        local now = tick()
        if now - lastHop > 8 then
            lastHop = now
            setStatus("no fruit, hopping")
            hop()
        end
    end
end

local function farmChests(hrp)
    local want = chestOrder[chestIdx]
    local chest
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name == want then
            chest = obj
            break
        end
    end
    if chest then
        tp(hrp, chest.Position)
        pcall(function() firetouchinterest(chest, hrp, 0) firetouchinterest(chest, hrp, 1) end)
        setStatus("chest " .. chest.Name)
    else
        setStatus("no " .. want)
    end
    chestIdx = (chestIdx % #chestOrder) + 1
end

local function getLevel()
    local ok, l = pcall(function() return LP:WaitForChild("Data"):WaitForChild("Level").Value end)
    return ok and l or 1
end

local function questNameForLevel(lvl)
    local best = QUESTS[1]
    for _, q in ipairs(QUESTS) do
        if not q.boss and q.lvl <= lvl then
            if q.lvl >= best.lvl then best = q end
        end
    end
    return best.name
end

local currentQuest = nil

local function takeQuest(force)
    pcall(function()
        local cf = ReplicatedStorage:FindFirstChild("Remotes")
        if cf then
            local comm = cf:FindFirstChild("CommF_")
            if comm then
                local name = questNameForLevel(getLevel())
                if force or name ~= currentQuest then
                    comm:InvokeServer("StartQuest", name, 1)
                    currentQuest = name
                end
            end
        end
    end)
end

local function finish(hrp)
    local enemies = Workspace:FindFirstChild("Enemies")
    if enemies then
        local list = {}
        for _, e in ipairs(enemies:GetChildren()) do
            local h = e:FindFirstChild("HumanoidRootPart")
            local hum = e:FindFirstChild("Humanoid")
            if h and hum and hum.Health > 0 then
                table.insert(list, e)
            end
        end
        if #list > 0 then
            local nearest = list[1]
            local nd = (nearest.HumanoidRootPart.Position - hrp.Position).Magnitude
            for _, e in ipairs(list) do
                local d = (e.HumanoidRootPart.Position - hrp.Position).Magnitude
                if d < nd then nd = d; nearest = e end
            end
            hoverAbove(hrp, nearest.HumanoidRootPart.Position)
            pcall(function() hrp.Velocity = Vector3.new(0, 0, 0) end)
            if (nearest.HumanoidRootPart.Position - hrp.Position).Magnitude <= auraRange then
                attack(nearest)
            end
            setStatus("killing " .. nearest.Name .. " (" .. #list .. " left) q:" .. tostring(currentQuest))
        else
            setStatus("quest clear -> next")
            currentQuest = nil
            takeQuest()
        end
    else
        setStatus("no enemies")
    end
end

task.spawn(function()
    while true do
        task.wait(0.3)
        local hrp = getHRP()
        if hrp then
            if State.BringFruit then bringFruits(hrp) end
            if State.FruitSniper then snipe(hrp) end
            if State.AutoFarmMoney then farmChests(hrp) end
            if State.AutoFinish then
                local now = tick()
                if now - lastQuest > 10 then lastQuest = now; takeQuest(true) end
                takeQuest()
                finish(hrp)
            end
        end
    end
end)
